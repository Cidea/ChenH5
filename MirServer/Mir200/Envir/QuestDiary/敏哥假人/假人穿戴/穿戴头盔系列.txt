[@拾取头盔触发]
{
#IF
CHECKVAR GLOBAL 假人_装备自换 = 开启
#ACT
#ELSEACT
BREAK

#IF
CHECKJOB warrior
CHECKTEXTLIST ..\QuestDiary\敏哥假人\假人穿戴\战士佩戴列表.txt <$STR(S$头盔名称)>
#ACT
DELAYGOTO 100 @战士替换头盔
BREAK

#IF
CHECKJOB wizard
CHECKTEXTLIST ..\QuestDiary\敏哥假人\假人穿戴\法师佩戴列表.txt <$STR(S$头盔名称)>
#ACT
DELAYGOTO 100 @法师替换头盔
BREAK

#IF
CHECKJOB taoist
CHECKTEXTLIST ..\QuestDiary\敏哥假人\假人穿戴\道士佩戴列表.txt <$STR(S$头盔名称)>
#ACT
DELAYGOTO 100 @道士替换头盔
BREAK
;-------------------------------------------------
[@战士替换头盔]
#IF
CHECKUSEITEM 4
#ACT
MOV N64 <$DC>
MOV M64 <$MAXDC>
MOV S$原始头盔 <$HELMET>
AutoTakeOnItem <$STR(S$头盔名称)> 4
#ELSEACT
AutoTakeOnItem <$STR(S$头盔名称)> 4
DELAYGOTO 1 @保存头盔数据
BREAK

#IF
LARGE <$MAXDC> M64
#ACT
DELAYGOTO 1 @保存头盔数据
BREAK

#IF
LARGE <$DC> N64
#ACT
DELAYGOTO 1 @保存头盔数据
BREAK

#ELSEACT
AutoTakeOnItem <$STR(S$原始头盔)> 4
BREAK

;-------------------------------------------------

[@法师替换头盔]
#IF
CHECKUSEITEM 4
#ACT
MOV N64 <$MC>
MOV M64 <$MAXMC>
MOV S$原始头盔 <$HELMET>
AutoTakeOnItem <$STR(S$头盔名称)> 4
#ELSEACT
AutoTakeOnItem <$STR(S$头盔名称)> 4
DELAYGOTO 1 @保存头盔数据
BREAK

#IF
LARGE <$MAXMC> M64
#ACT
DELAYGOTO 1 @保存头盔数据
BREAK

#IF
LARGE <$MC> N64
#ACT
DELAYGOTO 1 @保存头盔数据
BREAK

#ELSEACT
AutoTakeOnItem <$STR(S$原始头盔)> 4
BREAK

;-------------------------------------------------

[@道士替换头盔]
#IF
CHECKUSEITEM 4
#ACT
MOV N64 <$SC>
MOV M64 <$MAXSC>
MOV S$原始头盔 <$HELMET>
AutoTakeOnItem <$STR(S$头盔名称)> 4
#ELSEACT
AutoTakeOnItem <$STR(S$头盔名称)> 4
DELAYGOTO 1 @保存头盔数据
BREAK

#IF
LARGE <$MAXSC> M64
#ACT
DELAYGOTO 1 @保存头盔数据
BREAK

#IF
LARGE <$SC> N64
#ACT
DELAYGOTO 1 @保存头盔数据
BREAK

#ELSEACT
AutoTakeOnItem <$STR(S$原始头盔)> 4

[@保存头盔数据]
#IF
ISDUMMY
#ACT

AddTextListEx ..\QuestDiary\敏哥假人\假人配置\人物_<$HUMAN(假人_职业)>\<$USERNAME>.txt UseItems4=<$HELMET> 64
BREAK

}
