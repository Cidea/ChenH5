[@拾取盾牌触发]
{

#IF
CHECKJOB warrior
CHECKTEXTLIST ..\QuestDiary\一号假人\假人穿戴\战士佩戴列表.txt <$STR(S$盾牌名称)>
#ACT
DELAYGOTO 100 @战士替换盾牌

[@战士替换盾牌]
#IF

#ACT

BREAK

#IF
LARGE <$MAXDC> M76
#ACT
DELAYGOTO 1 @保存盾牌数据
BREAK

#IF
LARGE <$DC> N76
#ACT
DELAYGOTO 1 @保存盾牌数据

#ELSEACT

;-------------------------------------------------
