[@拾取勋章触发]
{

#IF
CHECKJOB warrior
CHECKTEXTLIST ..\QuestDiary\一号假人\假人穿戴\战士佩戴列表.txt <$STR(S$勋章名称)>
#ACT
DELAYGOTO 100 @战士替换勋章

#IF

#ACT

BREAK
;-------------------------------------------------
[@战士替换勋章]
#IF
CHECKUSEITEM 2
#ACT
MOV N62 <$DC>

#ELSEACT
AutoTakeOnItem <$STR(S$勋章名称)> 2

BREAK

#IF
LARGE <$MAXDC> M62
#ACT
DELAYGOTO 1 @保存勋章数据
BREAK

#IF
LARGE <$DC> N62
#ACT
DELAYGOTO 1 @保存勋章数据
BREAK

#ELSEACT

;-------------------------------------------------
