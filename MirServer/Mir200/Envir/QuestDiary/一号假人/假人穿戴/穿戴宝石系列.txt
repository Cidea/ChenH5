[@拾取宝石触发]
{

#IF
CHECKJOB warrior
CHECKTEXTLIST ..\QuestDiary\一号假人\假人穿戴\战士佩戴列表.txt <$STR(S$宝石名称)>
#ACT
DELAYGOTO 100 @战士替换宝石
BREAK

#IF
CHECKJOB wizard
CHECKTEXTLIST ..\QuestDiary\一号假人\假人穿戴\法师佩戴列表.txt <$STR(S$宝石名称)>
#ACT
DELAYGOTO 100 @法师替换宝石
BREAK

#IF
CHECKJOB taoist
CHECKTEXTLIST ..\QuestDiary\一号假人\假人穿戴\道士佩戴列表.txt <$STR(S$宝石名称)>
#ACT
DELAYGOTO 100 @道士替换宝石
BREAK
;-------------------------------------------------
[@战士替换宝石]
#IF
CHECKUSEITEM 12
#ACT
MOV N72 <$DC>
MOV M72 <$MAXDC>
MOV S$原始宝石 <$BOOTS>
AutoTakeOnItem <$STR(S$宝石名称)> 12
#ELSEACT
AutoTakeOnItem <$STR(S$宝石名称)> 12
DELAYGOTO 1 @保存宝石数据
BREAK

#IF
LARGE <$MAXDC> M72
#ACT
DELAYGOTO 1 @保存宝石数据
BREAK

#IF
LARGE <$DC> N72
#ACT
DELAYGOTO 1 @保存宝石数据
BREAK

#ELSEACT
AutoTakeOnItem <$STR(S$原始宝石)> 12

;-------------------------------------------------

[@法师替换宝石]
#IF
CHECKUSEITEM 12
#ACT

#IF

#ELSEACT

;-------------------------------------------------
